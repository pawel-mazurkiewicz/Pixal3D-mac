# Pixal3D Apple Silicon production environment — Python 3.10
#
# This is the pinned dependency set for the unified .venv-py310 (the
# unify-py310 consolidation: one Python 3.10 venv replacing the old 3.12
# Pixal3D venv + 3.11 trellis-mac bridge venv). Versions captured from a
# verified working environment that produces bit-parity meshes vs the prior
# 3.12 setup (fairy raw mesh V=3,940,419 / F=8,586,890).
#
# SETUP
#   /opt/homebrew/opt/python@3.10/bin/python3.10 -m venv .venv-py310
#   .venv-py310/bin/pip install -U pip wheel setuptools
#   uv pip install --python .venv-py310/bin/python -r requirements-mac.txt
#
#   # Vendored native Metal packages (compile xcrun .metal -> .metallib):
#   for p in mtlmesh mtlgemm mtlbvh mtldiffrast o_voxel; do
#     uv pip install --python .venv-py310/bin/python -e extern/$p --no-build-isolation
#   done
#
# NOTES
# - natten-mps==0.3.0 below is the COMMUNITY ssmall256 package (provides
#   natten_mps.compat.v020, imported by generate_mps.py). The in-house
#   extern/natten-mps (pawel-mazurkiewicz) is kept as source-of-truth but is
#   NOT installed from extern — see extern/README.md.
# - mtlgemm (flex_gemm) requires torch>=2.11 on Darwin for
#   at::mps::dispatch_sync_with_rethrow.
# - numpy is pinned <2.3 because 2.4+ dropped Python 3.10.
# - These are NOT the requirements-hfdemo.txt deps (that file is CUDA/Linux
#   only: cu124 index, nvdiffrast, flash_attn_3, CUDA wheels).
